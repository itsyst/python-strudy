# Uppgift 3:
assert isinstance(derivate(lambda x: x+1, 0.5), type(lambda x:x))

assert abs(derivate(lambda x: (x-1)**2, 0.1)(1)) < 0.0000000001
assert 1.99999999999 < abs(derivate(lambda x: (x)**2, 0.1)(1)) < 2.0000000001
